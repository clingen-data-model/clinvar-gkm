ClinVar-GKM 1.1.0 rebuild and republish: runbook¶

Status: prepared 2026-10-20, not executed. Every step that deploys, writes to BigQuery, GCS, R2 or Cloudflare, tags or merges is a separate step that the user confirms before it runs. Never touch live R2 without that confirmation. Never print the Cloudflare tokens.

Scope. Rebuild all 17 datasets with the 1.0 release branch (origin/1.0 @ 66e4af7), gate them, and publish them as 1.1.0 (ADR 0006, on PR #146) in the NCBI-style R2 layout (decided 2026-10-20, scripts in PR #182, ADR 0009 on #168): the current line v1/1.1/ gets 4 official monthlies (+ 00-latest), 3 weekly fulls for October (10-04, 10-11, 10-18, + 00-latest_weekly), 16 weekly deltas (October's at deltas/, July–September in deltas/YYYY-MM/) and one Parquet set at parquet/ built from 2026-10-18; the 1.0.1 generation moves from archive/1.0.1/ to the frozen line v1/1.0/; everything else in the bucket is deleted. #!/usr/bin/env python3
"""Validate EVERY record of a real published/staged bundle (.json.gz) with the GA4GH gkm-toolkit.

The bundle is streamed (ijson) so a 4.8 GB full never has to fit in memory. Records are passed to
ga4gh.gkm validate_bundle_schema VERBATIM (no cleanup, no re-collapse: unlike sub-bundle.py this cannot
mask an export/assembler bug), in chunks of --chunk records per section, each chunk wrapped as
{"schema_version": <bundle's own value>, <section>: {key: record, ...}}. The registry is built by
validate-schema-conformance.py's load_registry over the whole schema/ tree (strict, as published).

Also checks: the bundle's schema_version equals the schema const; every section is non-empty
(minProperties 1). Exit 0 only when every record passes.

Usage: python toolkit-validate-bundle.py <bundle.json.gz> <repo_root> [--workers N] [--chunk N] [--sections a,b]
"""
import argparse
import collections
import gzip
import importlib.util
import json
import multiprocessing as mp
import sys
import time
from pathlib import Path

import ijson

G = {}


def _init(repo_root):
    spec = importlib.util.spec_from_file_location(
        "vsc", Path(repo_root) / "src/scripts/validate-schema-conformance.py")
    vsc = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(vsc)
    from jsonschema import Draft202012Validator
    from ga4gh.gkm.bundles.schema_validation import validate_bundle_schema
    registry, _ = vsc.load_registry(Path(repo_root) / "schema", relax=False)
    schema = json.loads((Path(repo_root) / "schema/clinvar-gkm/clinvar-gkm-bundle.schema.json").read_text())
    G.update(schema=schema, validate=validate_bundle_schema,
             validator=Draft202012Validator(schema, registry=registry))


def _check(task):
    sv, section, items = task
    try:
        G["validate"]({"schema_version": sv, section: dict(items)}, G["schema"], validator=G["validator"])
        return section, len(items), []
    except Exception:  # find the failing records in this chunk
        bad = []
        for k, rec in items:
            try:
                G["validate"]({"schema_version": sv, section: {k: rec}}, G["schema"], validator=G["validator"])
            except Exception as e:  # noqa: BLE001
                bad.append(f"{section}/{k}: {str(e)[:300]}")
        if not bad:
            bad.append(f"{section}: chunk-level failure not reproducible per record")
        return section, len(items), bad


def records(path):
    """Yield ('schema_version', None, value) once, then (section, key, record) for every entry.

    Depth is tracked from the start/end events, NOT from dots in the ijson prefix: record keys such as
    'ga4gh:SL.x', 'clinvar.trait:1' or 'clinvar.submission:SCV1.2' contain dots (a dot-count depth
    silently dropped every such section)."""
    with gzip.open(path, "rb") as fh:
        lvl = 0; section = key = builder = None; bdepth = 0
        for prefix, event, value in ijson.parse(fh, use_float=True):
            if builder is not None:
                builder.event(event, value)
                if event in ("start_map", "start_array"):
                    bdepth += 1
                elif event in ("end_map", "end_array"):
                    bdepth -= 1
                    if bdepth == 0:
                        yield section, key, builder.value
                        builder = None
                continue
            if event in ("start_map", "start_array"):
                if lvl == 2:                       # a record value
                    builder = ijson.ObjectBuilder(); builder.event(event, value); bdepth = 1
                else:
                    lvl += 1
            elif event in ("end_map", "end_array"):
                lvl -= 1
            elif event == "map_key":
                if lvl == 1:
                    section = value
                elif lvl == 2:
                    key = value
            elif lvl == 1 and section == "schema_version":
                yield "schema_version", None, value
            elif lvl == 2:                         # scalar record value
                yield section, key, value


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("bundle")
    ap.add_argument("repo_root")
    ap.add_argument("--workers", type=int, default=max(1, mp.cpu_count() - 1))
    ap.add_argument("--chunk", type=int, default=2000)
    ap.add_argument("--max-print", type=int, default=20)
    ap.add_argument("--sections", default="", help="comma list: validate only these sections (others are counted only)")
    a = ap.parse_args()

    _init(a.repo_root)
    const = G["schema"]["properties"]["schema_version"].get("const")
    t0 = time.time()
    counts, failures, sv = {}, [], None
    only, skipped = set(filter(None, a.sections.split(","))), {}

    def tasks():
        nonlocal sv
        buf, cur = [], None
        for section, key, rec in records(a.bundle):
            if section == "schema_version":
                sv = rec
                continue
            if only and section not in only:
                skipped[section] = skipped.get(section, 0) + 1
                continue
            if cur is not None and section != cur and buf:
                yield sv, cur, buf
                buf = []
            cur = section
            buf.append((key, rec))
            if len(buf) >= a.chunk:
                yield sv, cur, buf
                buf = []
        if buf:
            yield sv, cur, buf

    nfail = 0
    with mp.Pool(a.workers, initializer=_init, initargs=(a.repo_root,)) as pool:
        pending = collections.deque()

        def drain(limit):
            nonlocal nfail
            while len(pending) > limit:  # bounded in-flight work: the bundle is never buffered in memory
                section, n, bad = pending.popleft().get()
                counts[section] = counts.get(section, 0) + n
                nfail += len(bad)
                failures.extend(bad[: max(0, 1000 - len(failures))])

        for task in tasks():
            pending.append(pool.apply_async(_check, (task,)))
            drain(a.workers * 3)
        drain(0)

    total = sum(counts.values())
    for s_, n_ in skipped.items():
        print(f"  (not validated: {s_} {n_:,d})")
    print(f"bundle: {a.bundle}")
    print(f"schema_version: bundle={sv!r} schema const={const!r} -> {'OK' if sv == const else 'MISMATCH'}")
    for s, n in counts.items():
        print(f"  {s:28s} {n:>10,d}")
    print(f"records: {total:,d}  failing: {nfail:,d}  ({time.time() - t0:,.0f} s, {a.workers} workers)")
    for f in failures[: a.max_print]:
        print("  FAIL", f)
    ok = sv == const and nfail == 0 and total > 0 and all(n > 0 for n in counts.values()) and not skipped
    print("PASS" if ok else "FAIL")
    sys.exit(0 if ok else 1)


if __name__ == "__main__":
    main()

  SELECT job_id, creation_time, REGEXP_EXTRACT(query, r'clinvar_ingest\.([a-z_]+_incremental)') proc
  FROM \`region-us\`.INFORMATION_SCHEMA.JOBS_BY_PROJECT
  WHERE creation_time > ${since_sql} AND parent_job_id IS NULL AND error_result IS NULL
    AND REGEXP_CONTAINS(query, r\"_proc_incremental\`\\('${d}'\")),
last AS (SELECT * FROM p QUALIFY ROW_NUMBER() OVER (PARTITION BY proc ORDER BY creation_time DESC) = 1)
SELECT l.proc, COUNTIF(STRPOS(c.query, '${base}') > 0 AND NOT REGEXP_CONTAINS(c.query, r'INFORMATION_SCHEMA|gkm_pipeline_version')) AS base_reads
FROM last l JOIN \`region-us\`.INFORMATION_SCHEMA.JOBS_BY_PROJECT c ON c.parent_job_id = l.job_id
WHERE c.creation_time > ${since_sql}
GROUP BY 1 ORDER BY 1")"
echo "$out" | sed 1d | sed "s/^/  $d /"
n_ok="$(echo "$out" | sed 1d | awk -F, '$2 > 0' | wc -l | tr -d ' ')"
n_all="$(echo "$out" | sed 1d | wc -l | tr -d ' ')"
if [ "$n_all" = 7 ] && [ "$n_ok" = 7 ]; then echo "INCREMENTAL $d: 7/7 procs read baseline $base"
else echo "NOT INCREMENTAL $d: $n_ok/$n_all procs read baseline $base (fell back to full, or a proc did not run)"; exit 1; fi
