Area 1 — Variant, the subject of every proposition
Query — drill one variant through VRS (members[] → allele → location → sequenceReference) 
-- members[] holds VRS ids; the VA. prefix dispatches to allele
SELECT v.id AS variation, v.name AS variation_name,
       a.id AS allele, l.id AS location, sr.id AS seqref
FROM 'variation.parquet' v, unnest(v.members) AS t(member)
LEFT JOIN 'allele.parquet'            a  ON a.id  = member
LEFT JOIN 'location.parquet'          l  ON l.id  = a.location_id
LEFT JOIN 'sequenceReference.parquet' sr ON sr.id = l.sequence_reference_id
WHERE v.id = 'clinvar:51592';

Area 2 — Condition & therapy, the object
Query — expand a conditionSet to member names (concepts[] holds ids, not names) 
-- concepts[] holds ids; join to condition, then join the names with ';'
SELECT cs.id, cs.membershipOperator AS op, string_agg(c.name, '; ') AS members
FROM 'conditionSet.parquet' cs, unnest(cs.concepts) AS t(cid)
LEFT JOIN 'condition.parquet' c ON c.id = cid
WHERE cs.id = 'clinvar.conditionset:vcv-50aae75c5e3a7740ab0c2f3ff9ebd970'
GROUP BY cs.id, cs.membershipOperator;

Area 3 — Proposition, what is being claimed
Query — every condition a variant is asserted for (object ids resolved to names) 
-- resolve each object to a name: condition.name, or the set's members joined by ';'
WITH set_names AS (
    SELECT cs.id, string_agg(c.name, '; ') AS member_names
    FROM 'conditionSet.parquet' cs, unnest(cs.concepts) AS t(cid)
    LEFT JOIN 'condition.parquet' c ON c.id = cid GROUP BY cs.id
)
SELECT DISTINCT p.object_condition_id, coalesce(c.name, sn.member_names) AS condition
FROM 'varcond-proposition.parquet' p
LEFT JOIN 'condition.parquet' c  ON c.id  = p.object_condition_id
LEFT JOIN set_names           sn ON sn.id = p.object_condition_id
WHERE p.subject_variant_id = 'clinvar:51592'            -- NM_000059.4(BRCA2):c.4090A>C
  AND coalesce(c.name, sn.member_names) IS NOT NULL
ORDER BY condition LIMIT 12;

Area 4 — SCV, one submitter's assertion
Query — an SCV's submitter and classification 
-- contributions[].contributor is the submitter id
SELECT DISTINCT s.classification.name AS classification, sub.name AS submitter
FROM 'scv.parquet' s, unnest(s.contributions) AS t(con)
JOIN 'submitter.parquet' sub ON sub.id = con.contributor
WHERE s.proposition_id IN (
    SELECT id FROM 'varcond-proposition.parquet' WHERE subject_variant_id = 'clinvar:51592')
LIMIT 12;
