## Appendix A — `toolkit-validate-bundle.py`

Write to `$RB/tools/toolkit-validate-bundle.py`. Tested on the live 2026-1011 delta (streams, bounded memory,
reports per-record failures).

```python
#!/usr/bin/env python3
"""Validate EVERY record of a real published/staged bundle (.json.gz) with the GA4GH gkm-toolkit.

The bundle is streamed (ijson) so a 4.8 GB full never has to fit in memory. Records are passed to
ga4gh.gkm validate_bundle_schema VERBATIM (no cleanup, no re-collapse: unlike sub-bundle.py this cannot
mask an export/assembler bug), in chunks of --chunk records per section, each chunk wrapped as
{"schema_version": <bundle's own value>, <section>: {key: record, ...}}. The registry is built by
validate-schema-conformance.py's load_registry over the whole schema/ tree (strict, as published).

Also checks: the bundle's schema_version equals the schema const; every section is non-empty
(minProperties 1). Exit 0 only when every record passes.

Usage: python toolkit-validate-bundle.py <bundle.json.gz> <repo_root> [--workers N] [--chunk N] [--sections a,b]
"""
import argparse
import collections
import gzip
import importlib.util
import json
import multiprocessing as mp
import sys
import time
from pathlib import Path

import ijson

G = {}


def _init(repo_root):
    spec = importlib.util.spec_from_file_location(
        "vsc", Path(repo_root) / "src/scripts/validate-schema-conformance.py")
    vsc = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(vsc)
    from jsonschema import Draft202012Validator
    from ga4gh.gkm.bundles.schema_validation import validate_bundle_schema
    registry, _ = vsc.load_registry(Path(repo_root) / "schema", relax=False)
    schema = json.loads((Path(repo_root) / "schema/clinvar-gkm/clinvar-gkm-bundle.schema.json").read_text())
    G.update(schema=schema, validate=validate_bundle_schema,
             validator=Draft202012Validator(schema, registry=registry))


def _check(task):
    sv, section, items = task
    try:
        G["validate"]({"schema_version": sv, section: dict(items)}, G["schema"], validator=G["validator"])
        return section, len(items), []
    except Exception:  # find the failing records in this chunk
        bad = []
        for k, rec in items:
            try:
                G["validate"]({"schema_version": sv, section: {k: rec}}, G["schema"], validator=G["validator"])
            except Exception as e:  # noqa: BLE001
                bad.append(f"{section}/{k}: {str(e)[:300]}")
        if not bad:
            bad.append(f"{section}: chunk-level failure not reproducible per record")
        return section, len(items), bad


def records(path):
    """Yield ('schema_version', value) once, then (section, key, record) for every entry."""
    with gzip.open(path, "rb") as fh:
        events = ijson.parse(fh, use_float=True)
        section = key = builder = None
        for prefix, event, value in events:
            depth = prefix.count(".") + 1 if prefix else 0
            if builder is not None:
                builder.event(event, value)
                if event in ("end_map", "end_array") and prefix == f"{section}.{key}":
                    yield section, key, builder.value
                    builder = None
                continue
            if depth == 0 and event == "map_key":
                section = value
            elif prefix == "schema_version" and event in ("string", "number"):
                yield "schema_version", None, value
            elif depth == 1 and event == "map_key":
                key = value
            elif depth == 2 and event in ("start_map", "start_array") and prefix == f"{section}.{key}":
                builder = ijson.ObjectBuilder()
                builder.event(event, value)
            elif depth == 2 and prefix == f"{section}.{key}":   # scalar entry value
                yield section, key, value


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("bundle")
    ap.add_argument("repo_root")
    ap.add_argument("--workers", type=int, default=max(1, mp.cpu_count() - 1))
    ap.add_argument("--chunk", type=int, default=2000)
    ap.add_argument("--max-print", type=int, default=20)
    ap.add_argument("--sections", default="", help="comma list: validate only these sections (others are counted only)")
    a = ap.parse_args()

    _init(a.repo_root)
    const = G["schema"]["properties"]["schema_version"].get("const")
    t0 = time.time()
    counts, failures, sv = {}, [], None
    only, skipped = set(filter(None, a.sections.split(","))), {}

    def tasks():
        nonlocal sv
        buf, cur = [], None
        for section, key, rec in records(a.bundle):
            if section == "schema_version":
                sv = rec
                continue
            if only and section not in only:
                skipped[section] = skipped.get(section, 0) + 1
                continue
            if cur is not None and section != cur and buf:
                yield sv, cur, buf
                buf = []
            cur = section
            buf.append((key, rec))
            if len(buf) >= a.chunk:
                yield sv, cur, buf
                buf = []
        if buf:
            yield sv, cur, buf

    nfail = 0
    with mp.Pool(a.workers, initializer=_init, initargs=(a.repo_root,)) as pool:
        pending = collections.deque()

        def drain(limit):
            nonlocal nfail
            while len(pending) > limit:  # bounded in-flight work: the bundle is never buffered in memory
                section, n, bad = pending.popleft().get()
                counts[section] = counts.get(section, 0) + n
                nfail += len(bad)
                failures.extend(bad[: max(0, 1000 - len(failures))])

        for task in tasks():
            pending.append(pool.apply_async(_check, (task,)))
            drain(a.workers * 3)
        drain(0)

    total = sum(counts.values())
    for s_, n_ in skipped.items():
        print(f"  (not validated: {s_} {n_:,d})")
    print(f"bundle: {a.bundle}")
    print(f"schema_version: bundle={sv!r} schema const={const!r} -> {'OK' if sv == const else 'MISMATCH'}")
    for s, n in counts.items():
        print(f"  {s:28s} {n:>10,d}")
    print(f"records: {total:,d}  failing: {nfail:,d}  ({time.time() - t0:,.0f} s, {a.workers} workers)")
    for f in failures[: a.max_print]:
        print("  FAIL", f)
    ok = sv == const and nfail == 0 and total > 0 and all(n > 0 for n in counts.values()) and not skipped
    print("PASS" if ok else "FAIL")
    sys.exit(0 if ok else 1)


if __name__ == "__main__":
    main()
```

## Appendix B — `bundle-parquet-check.py`

```python
#!/usr/bin/env python3
"""Bundle <-> Parquet gate: the Parquet set must be recomputable from the bundle (ADR 0008).

For every Parquet table (one file per section; evidenceLine + vcv_evidenceLine + rcv_evidenceLine all map to
the bundle's evidenceLine section; therapy/therapyGroup use key/value, the rest id/data):

Usage: bundle-parquet-check.py <bundle.json.gz> <parquet_dir_or_url_prefix> [--mod 500] [--tables a,b,...]
       [--subset]   bundle is a delta: only check bundle -> Parquet (ids present, sampled data equal)
Exit 0 only when every check passes.

import argparse
import gzip
import json
import os
import sys
import tempfile
import zlib

import duckdb
import ijson

EVIDENCE = {"evidenceLine", "vcv_evidenceLine", "rcv_evidenceLine"}


def canon(v):
    return json.dumps(v, sort_keys=True, separators=(",", ":"), ensure_ascii=False)


def stream(path):

        section = key = builder = None
        for prefix, event, value in ijson.parse(fh, use_float=True):

                key = value
            elif depth == 2 and prefix == f"{section}.{key}":
                if event in ("start_map", "start_array"):
                    builder = ijson.ObjectBuilder()
                    builder.event(event, value)
                else:
                    yield section, key, value


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("bundle")
    ap.add_argument("parquet")
    ap.add_argument("--mod", type=int, default=500)
    ap.add_argument("--tables", default="")
    ap.add_argument("--subset", action="store_true")
    a = ap.parse_args()

    remote = a.parquet.startswith("http")
    tables = [t for t in a.tables.split(",") if t] or sorted(
        f[:-8] for f in os.listdir(a.parquet) if f.endswith(".parquet"))
    sec_of = {t: ("evidenceLine" if t in EVIDENCE else t) for t in tables}
    wanted = set(sec_of.values())

    tmp = tempfile.mkdtemp(prefix="bpq-")
    ids_path = os.path.join(tmp, "bundle_ids.tsv")
    sample, counts = {}, {}
    with open(ids_path, "w") as ids:
        for section, key, rec in stream(a.bundle):
            if section == "schema_version" or section not in wanted:
                continue
            counts[section] = counts.get(section, 0) + 1
            ids.write(f"{section}\t{key}\n")
            if zlib.crc32(key.encode()) % a.mod == 0:
                sample[(section, key)] = canon(rec)

    con = duckdb.connect()
    if remote:
        con.execute("INSTALL httpfs; LOAD httpfs;")
    con.execute(f"CREATE TABLE b AS SELECT * FROM read_csv('{ids_path}', delim='\t', header=false, "
                "columns={'section':'VARCHAR','k':'VARCHAR'}, quote='', escape='')")
    srcs = []
    for t in tables:
        url = f"{a.parquet.rstrip('/')}/{t}.parquet"
        cols = [r[0] for r in con.execute(f"DESCRIBE SELECT * FROM read_parquet('{url}')").fetchall()]
        idc, dc = ("id", "data") if "id" in cols else ("key", "value")
        srcs.append(f"SELECT '{sec_of[t]}' AS section, '{t}' AS tbl, {idc}::VARCHAR AS k, {dc}::VARCHAR AS d "
                    f"FROM read_parquet('{url}')")
    con.execute("CREATE TABLE p AS " + " UNION ALL ".join(srcs))

    fail = False
    print(f"{'section':30s} {'bundle':>11s} {'parquet':>11s} {'b-not-p':>9s} {'p-not-b':>9s}")
    for (section,) in con.execute("SELECT DISTINCT section FROM p ORDER BY 1").fetchall():
        pn = con.execute("SELECT COUNT(*) FROM p WHERE section=?", [section]).fetchone()[0]
        bn = counts.get(section, 0)
        b_not_p = con.execute("SELECT COUNT(*) FROM b WHERE section=? AND k NOT IN "
                              "(SELECT k FROM p WHERE section=?)", [section, section]).fetchone()[0]
        p_not_b = 0 if a.subset else con.execute(
            "SELECT COUNT(*) FROM p WHERE section=? AND k NOT IN (SELECT k FROM b WHERE section=?)",
            [section, section]).fetchone()[0]
        dup = con.execute("SELECT COUNT(*) - COUNT(DISTINCT k) FROM p WHERE section=?", [section]).fetchone()[0]
        bad = b_not_p or p_not_b or dup or (not a.subset and pn != bn)
        fail |= bool(bad)
        print(f"{section:30s} {bn:>11,d} {pn:>11,d} {b_not_p:>9,d} {p_not_b:>9,d}"
              f"{'  dup-ids=' + str(dup) if dup else ''}{'  <-- FAIL' if bad else ''}")

    con.execute("CREATE TABLE s (section VARCHAR, k VARCHAR)")
    con.executemany("INSERT INTO s VALUES (?, ?)", list(sample.keys()))
    got = {(s, k): d for s, k, d in con.execute(
        "SELECT p.section, p.k, p.d FROM p JOIN s USING (section, k)").fetchall()}
    mism = [sk for sk, j in sample.items() if sk not in got or canon(json.loads(got[sk])) != j]
    print(f"data spot-check: {len(sample):,d} sampled records (crc32 % {a.mod} == 0), {len(mism):,d} differ")
    for s, k in mism[:10]:
        print(f"  DIFF {s}/{k}")
    fail |= bool(mism) or not sample
    print("FAIL" if fail else "PASS")
    sys.exit(1 if fail else 0)


if __name__ == "__main__":
    main()
```

## Appendix C — `expected_diff.sh`

Write to `$RB/tools/expected_diff.sh` and `chmod +x` it. Usage: `expected_diff.sh <dataset>` (compares
`<t>__pre110` → `<t>`); `TABLES=` limits the tables; `OLD_DS=… expected_diff.sh <ds> "" ""` compares two datasets.
Dry-run (`expected_diff.sh <ds> --dry-run`) passes on all 21 tables (10-04 → 10-11: ~79 GB).

```bash
#!/bin/bash
# expected-diff gate: classify every row that differs between <ds>.<t>__pre110 and <ds>.<t>.
# Usage: expected_diff.sh <dataset> [--dry-run] [OLD_SUFFIX]   (OLD_SUFFIX default __pre110)
set -euo pipefail
DS="$1"; DRY="${2:-}"; SUF="${3-__pre110}"
OLD_DS="${OLD_DS:-$DS}"
PTR="r'\"#/(?:var[a-z]+-proposition|proposition[A-Z][A-Za-z]*)/'"
ALL="gkm_dict_sequence_reference gkm_dict_location gkm_dict_allele gkm_dict_copy_number_count \
         gkm_dict_copy_number_change gkm_dict_gene gkm_dict_submitter gkm_dict_therapy gkm_dict_therapygroup \
         gkm_dict_proposition gkm_dict_vcv_proposition gkm_dict_rcv_proposition \
         gkm_dict_variation gkm_dict_condition gkm_dict_condition_set \
         gkm_dict_evidence_line gkm_dict_vcv_evidence_line gkm_dict_rcv_evidence_line \
         gkm_dict_scv gkm_dict_vcv gkm_dict_rcv"
for T in ${TABLES:-$ALL}; do
  case "$T" in
    gkm_dict_variation|gkm_dict_condition|gkm_dict_condition_set|*evidence_line|gkm_dict_scv|gkm_dict_vcv|gkm_dict_rcv)
      K=id; J=t ;;
    *) K=key; J=t.value ;;
  esac
  bq query --quiet --use_legacy_sql=false --format=csv --max_rows=1000 ${DRY:+--dry_run} "
    WITH o AS (SELECT CAST(t.${K} AS STRING) pk, TO_JSON_STRING(${J}) j FROM \`${OLD_DS}.${T}${SUF}\` t),
         n AS (SELECT CAST(t.${K} AS STRING) pk, TO_JSON_STRING(${J}) j FROM \`${DS}.${T}\` t),
         d AS (SELECT pk, o.j oj, n.j nj FROM o FULL JOIN n USING (pk) WHERE o.j IS DISTINCT FROM n.j),
         c AS (SELECT pk, oj, nj,
                 REGEXP_REPLACE(oj, ${PTR}, '\"#/P/') ojn, REGEXP_REPLACE(nj, ${PTR}, '\"#/P/') njn FROM d)
    SELECT '${T}' AS table_name,
      CASE WHEN oj IS NULL THEN 'ADDED' WHEN nj IS NULL THEN 'REMOVED'
           WHEN ojn = njn THEN 'pointer-section-only'
           ELSE IFNULL(NULLIF(\`clinvar_ingest.json_changed_keys\`(ojn, njn), ''), '(order-only)') END AS change,
      COUNT(*) AS n, MIN(pk) AS example_pk
    FROM c GROUP BY 1, 2 ORDER BY n DESC" | { [[ -n "$DRY" ]] && sed "s|^|${T}: |" || tail -n +2; }
done
```
